Comprobación Par / Impar

let numPedido = prompt("Introduce un número: "); let numero = Number(numPedido); if(numero === null || numero.trim === "" || isNaN(numero)){ console.warn("Lo que has introducido es incorrecto") } if(numPedido % 2 === 0){ console.log("El número " + numPedido + " es par") }else{ console.log("El número " + numPedido + " es impar") }
